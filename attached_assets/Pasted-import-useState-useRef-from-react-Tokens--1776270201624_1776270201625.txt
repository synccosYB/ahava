// ── Toast ─────────────────────────────────────────────────────────────────────
let _toastFn=()=>{};
const addToast=(msg,type="success")=>_toastFn(msg,type);

function ToastHost(){
  const [toasts,setToasts]=useState([]);
  _toastFn=(msg,type)=>{
    const id=Date.now();
    setToasts(p=>[...p,{id,msg,type}]);
    setTimeout(()=>setToasts(p=>p.filter(t=>t.id!==id)),3400);
  };
  return(
    <div style={{position:"fixed",bottom:24,right:24,zIndex:9999,display:"flex",flexDirection:"column",gap:10,pointerEvents:"none"}}>
      {toasts.map(t=>(
        <div key={t.id} style={{background:t.type==="error"?C.red:t.type==="warning"?C.amber:C.green,color:"#fff",borderRadius:10,padding:"12px 18px",fontSize:13,fontWeight:600,boxShadow:"0 4px 16px rgba(0,0,0,0.18)",maxWidth:320,display:"flex",gap:10,alignItems:"center"}}>
          <span>{t.type==="error"?"❌":t.type==="warning"?"⚠️":"✅"}</span>{t.msg}
        </div>
      ))}
    </div>
  );
}

function PTOBar({used,pending,total,color}){
  const usedPct=(used/total)*100;
  const pendPct=(pending/total)*100;
  const left=total-used-pending;
  return(
    <div>
      <div style={{height:8,borderRadius:99,background:"#e5e7eb",overflow:"hidden",display:"flex"}}>
        <div style={{width:`${usedPct}%`,background:color,borderRadius:99}}/>
        <div style={{width:`${pendPct}%`,background:color+"66"}}/>
      </div>
      <div style={{display:"flex",gap:16,marginTop:5,fontSize:11,color:C.gray}}>
        <span><span style={{color,fontWeight:700}}>{used}</span> used</span>
        {pending>0&&<span><span style={{color:color+"99",fontWeight:700}}>{pending}</span> pending</span>}
        <span><span style={{fontWeight:700,color:C.text}}>{left}</span> remaining</span>
        <span style={{marginLeft:"auto"}}>{total} total</span>
      </div>
    </div>
  );
}
